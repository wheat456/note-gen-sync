1. 类名建议以 `Extension` 结尾（XAML 中可以省略）
# 无参数
```cs

public class MyCustomExtension : MarkupExtension
{
    // 2. 重写核心方法：返回你要给 XAML 的值
    public override object ProvideValue(IServiceProvider serviceProvider)
    {
        // 这里写你的逻辑
        return "我是自定义标记扩展的值";
    }
}

```

# 有参数
```cs
public class FormatTextExtension : MarkupExtension
{
    // 定义参数（公开属性）
    public string Message { get; set; }
    public int Number { get; set; }

    // 核心方法
    public override object ProvideValue(IServiceProvider serviceProvider)
    {
        // 自定义逻辑：拼接字符串
        return $"消息：{Message}，数字：{Number}";
    }
}
```

- 使用
- ```xml
  <!-- 写法1：键值对（推荐，清晰） -->
<TextBlock Text="{FormatText Message=你好, Number=100}" />

<!-- 写法2：位置参数（需要加构造函数） -->
<TextBlock Text="{FormatText 你好, 100}" />
  
  ```

# 获取上下文

```cs
public class TargetInfoExtension : MarkupExtension
{
    public override object ProvideValue(IServiceProvider serviceProvider)
    {
        // 1. 获取目标对象服务
        var provide = serviceProvider.GetService(typeof(IProvideValueTarget)) as IProvideValueTarget;
        
        if (provide == null) return null;

        // 2. 获取使用这个扩展的控件（目标对象）
        var targetControl = provide.TargetObject as System.Windows.FrameworkElement;
        // 3. 获取使用的属性（比如 Text、Content）
        var targetProperty = provide.TargetProperty;

        // 4. 可以拿到控件的 DataContext
        var viewModel = targetControl?.DataContext;

        return $"控件：{targetControl?.Name}，属性：{targetProperty?.Name}";
    }
}
```
