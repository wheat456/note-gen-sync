- **`BackgroundService`** 是一个方便使用的**抽象基类**，它已经为你实现了 `IHostedService` 接口。它最大的优势在于它将复杂的 `StartAsync` 和 `StopAsync` 封装起来，让你只需要专注于实现一个核心方法。
## 任务
```cs
public class MyBackgroundService : BackgroundService
{
    // ... 构造函数和配置注入

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        // 循环执行任务，直到收到停止信号
        while (!stoppingToken.IsCancellationRequested) 
        {
            // 1. 执行你的后台任务
            Console.WriteLine("Worker running at: " + DateTimeOffset.Now);

            // 2. 等待一段时间，同时监听停止信号
            await Task.Delay(5000, stoppingToken); 
        }

        Console.WriteLine("Worker is shutting down gracefully...");
    }
}
```

# logger 
```cs
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.Hosting;

public class MyBackgroundService : BackgroundService
{
    private readonly ILogger<MyBackgroundService> _logger; // 注入 ILogger<T>

    public MyBackgroundService(ILogger<MyBackgroundService> logger) 
    {
        _logger = logger;
    }
    
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        _logger.LogInformation("MyBackgroundService is starting its work."); // 使用日志方法

        while (!stoppingToken.IsCancellationRequested) 
        {
            _logger.LogWarning("Worker running at: {Time}", DateTimeOffset.Now);

            await Task.Delay(5000, stoppingToken); 
        }

        _logger.LogInformation("MyBackgroundService has shut down gracefully.");
    }
}
```
