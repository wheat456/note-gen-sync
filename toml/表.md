# 不能重复定义表
```toml
# 不要这样做

[fruit]
apple = "红"

[fruit]
orange = "橙"
```

```toml
# 也不要这样做

[fruit]
apple = "红"

[fruit.apple]
texture = "光滑"
```
```toml
[fruit]
apple.color = "红"
apple.taste.sweet = true

# [fruit.apple]  # 非法
# [fruit.apple.taste]  # 非法

[fruit.apple.texture]  # 你可以添加子表
smooth = true
```


# 不要无序定义表
```toml
# 有效但不鼓励
[fruit.apple]
[animal]
[fruit.orange]
```
```toml
# 推荐
[fruit.apple]
[fruit.orange]
[animal]
```
# 顶层表
```toml
# 顶层表开始。
name = "Fido"
breed = "pug"

# 顶层表结束。
[owner]
name = "Regina Dogman"
member_since = 1999-08-04
```
