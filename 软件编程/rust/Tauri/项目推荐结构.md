```rs
my-tauri-app/
├── src/                      # 前端源代码
└── src-tauri/                # 后端核心目录
    ├── Cargo.toml            # Rust 依赖清单与项目配置
    ├── Cargo.lock            # 依赖版本锁定文件
    ├── tauri.conf.json       # Tauri 核心配置 (窗口、安全、构建配置)
    ├── build.rs              # 构建脚本 (负责编译期 Tauri 上下文注入)
    ├── capabilities/         # [Tauri v2] 权限与安全策略配置 (JSON)
    ├── icons/                # 应用图标资产 (.ico, .png, .icns)
    └── src/                  # Rust 源码目录
        ├── main.rs           # 桌面端入口 (或 Tauri v1 单一入口)
        ├── lib.rs            # 核心库入口 (Tauri v2 移动端与桌面端共享逻辑)
        ├── commands/         # IPC 接口层 (暴露给前端调用的命令)
        │   ├── mod.rs
        │   ├── auth.rs
        │   └── system.rs
        ├── state/            # 全局状态层 (Tauri State 管理)
        │   ├── mod.rs
        │   └── app_state.rs
        ├── models/           # 数据模型层 (与前端交互的 Serde 结构体)
        │   ├── mod.rs
        │   └── user.rs
        ├── services/         # 业务逻辑层 (数据库、网络、文件 IO)
        │   ├── mod.rs
        │   └── db_service.rs
        └── utils/            # 工具与错误处理层
            ├── mod.rs
            └── error.rs
```
