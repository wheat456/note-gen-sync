## 多重绑定
- 绑定的值依赖多个属性的计算结果

```xml
<TextBlock Text="MultiBinding Text Color!" FontSize="24">  
<TextBlock.Foreground>  
<MultiBinding Converter="{StaticResource RgbToBrushMultiConverter}">  
<Binding Path="Value" ElementName="red" />  
<Binding Path="Value" ElementName="green" />  
<Binding Path="Value" ElementName="blue" />  
</MultiBinding>  
</TextBlock.Foreground>

```


```cs
public sealed class RgbToBrushMultiConverter : IMultiValueConverter
{
    public object? Convert(IList<object?> values, Type targetType, object? parameter, CultureInfo culture)
    {
        // 确保提供了所有绑定，并且绑定到了正确的目标类型
        if (values?.Count != 3 || !targetType.IsAssignableFrom(typeof(ImmutableSolidColorBrush)))
            throw new NotSupportedException();

        // 确保所有绑定都是正确的类型
        if (!values.All(x => x is decimal or UnsetValueType or null))
            throw new NotSupportedException();

        // 提取值，如果有任何值未设置则返回 DoNothing。
        // 在绑定初始化期间，Convert会被多次调用，
        // 因此一些属性最初会是未设置的状态。
        if (values[0] is not decimal r ||
            values[1] is not decimal g ||
            values[2] is not decimal b)
            return BindingOperations.DoNothing;

        byte a = 255;
        var color = new Color(a, (byte)r, (byte)g, (byte)b);
        return new ImmutableSolidColorBrush(color);
    }
}
```


# 属性
# 直接属性
直接属性**不能**在 `<Style>` 的 `<Setter>` 里使用。

```cs

using Avalonia;
using Avalonia.Controls;
using Avalonia.Data;

namespace YourProject.Controls;

public class MyCustomControl : Control // 也可以是 TemplatedControl 或 UserControl
{
    // --- 第一步：定义私有后端存储字段 ---
    private string? _myProperty;

    // --- 第二步：注册直接属性 (必须是 static readonly) ---
    public static readonly DirectProperty<MyCustomControl, string?> MyPropertyProperty =
        AvaloniaProperty.RegisterDirect<MyCustomControl, string?>(
            nameof(MyProperty),           // 属性的名称
            o => o.MyProperty,            // Getter：告诉框架如何读这个值
            (o, v) => o.MyProperty = v,   // Setter：告诉框架如何写这个值
            defaultBindingMode: BindingMode.TwoWay // (可选) 设置默认绑定模式
        );

    // --- 第三步：包装成标准的 C# 公开属性 ---
    public string? MyProperty
    {
        get => _myProperty;
        // --- 第四步：使用 SetAndRaise 通知 UI 更新 ---
        set => SetAndRaise(MyPropertyProperty, ref _myProperty, value);
    }
}
```

# 样式属性
```cs
using Avalonia;
using Avalonia.Controls;

namespace YourProject.Controls;

public class MyCustomControl : Control
{
    // --- 第一步：注册并定义属性标识符 (必须是 static readonly) ---
    public static readonly StyledProperty<string?> MyStringProperty =
        AvaloniaProperty.Register<MyCustomControl, string?>(
            nameof(MyString),           // 属性名
            defaultValue: "默认值",      // 可选：设置初始默认值
            inherits: false             // 可选：是否允许子控件继承此值
        );

    // --- 第二步：包装成标准的 C# 属性 ---
    public string? MyString
    {
        get => GetValue(MyStringProperty);
        set => SetValue(MyStringProperty, value);
    }
}
```

# 依赖属性

```cs
using Avalonia;
using Avalonia.Data;

namespace YourProject.Helper;

public static class MyProperties
{
    // 1. 注册附加属性 (Attached Property)
    // 泛型参数：宿主类型(这里用Control表示通用), 属性类型
    //control 是组件类型，string是附加属性的类型
    public static readonly AttachedProperty<string?> RemarkProperty =
        Avalonia.AvaloniaProperty.RegisterAttached<Control, string?>(
            "Remark",           // 属性名
            typeof(MyProperties) // 哪个类定义的
        );

    // 2. 提供 Accessor (Getter) - 必须是静态的
    // Getxxx 命名规则
    // Control 控件类型
    public static string? GetRemark(Control element) => element.GetValue(RemarkProperty);

    // 3. 提供 Accessor (Setter) - 必须是静态的
    public static void SetRemark(Control element, string? value) => element.SetValue(RemarkProperty, value);
}

```
