effect中传入一个无参函数
可以多写effect实现不同作用方向
- **`$effect`**：95% 场景的首选。数据变了 $\rightarrow$ **DOM 渲染完了** $\rightarrow$ 执行它。
- **`$effect.pre`**：仅在操作 DOM 几何尺寸/滚动条时使用。数据变了 $\rightarrow$ **趁 DOM 还没渲染** $\rightarrow$ 执行它
```ts
<script>
    // ========================================================
    // 核心生命周期与状态追踪：$effect 块
    // ========================================================
    $effect(() => {
        
        // --------------------------------------------------------
        // 【A. 挂载时加载】 & 【C. 状态变了后运行】
        // --------------------------------------------------------
        // 触发时机 1（挂载）：当这个组件第一次出现在屏幕上时，这里会执行。
        // 触发时机 2（状态变化）：因为里面读取了 [roomId]，以后每次 roomId 改变，这里都会重新执行！
        console.log(`【连接中】正在连接到：${roomId}`);
        
        // 模拟建立 WebSocket 连接
        const connection = connectToChat(roomId); 


        // --------------------------------------------------------
        // 【B. 卸载时运行】（通过 return 一个清理函数）
        // --------------------------------------------------------
        return () => {
            // 触发时机 1（状态变化前的清理）：在由于 roomId 改变准备重新跑上面的代码前，
            //                              为了防止连接重复，会先跑这里把旧的连接断开。
            // 触发时机 2（卸载）：当这个组件彻底从屏幕上消失（组件销毁）时，这里会执行最后一次。
            console.log(`【断开连接】已离开：${roomId}`);
            
            connection.disconnect(); 
        };
    });

    // 模拟的伪代码函数，不用深究
    function connectToChat(id) {
        return { disconnect: () => console.log(`彻底销毁了对 ${id} 的物理连接`) };
    }
</script>
```
