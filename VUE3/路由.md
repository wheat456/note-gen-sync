在`views`或者`pages`中写组件，比如`home.vue、car.vue`
在router文件夹中，写index.ts
```ts
import {createRouter,createWebHistory} from 'vue-router'

import Fisrt from '../components/Fisrt.vue'

import Second from '../components/Second.vue'

// Make sure Third.vue exists in src/components, or correct the path if necessary

import Third from '../components/Third.vue'
  
const  router = createRouter({

    history:createWebHistory(),





})
export default router
```
创建路由
在app.vue中引入路由
```ts
<script setup lang="ts">

import { ref, computed } from 'vue'
import { RouterLink, RouterView } from 'vue-router'
 
</script>
 
<template>
  <div>
    <nav>
      <RouterLink to="/first">第一页</RouterLink>
      <RouterLink to="/second">这是第二页</RouterLink>
      <RouterLink to="/third">这是第三页</RouterLink>
    </nav>
    <RouterView />
  </div>
</template>
```

在**main.ts**中
```ts
import { createApp } from 'vue'
import App from './App.vue'
import router from './router'
const app = createApp(App)
app.use(router)
app.mount('#app')
```

# 工作模式
### history
不带#号
### hash
路径中有#号，服务端不用处理路径问题

# to

```ts
<RouterLink to="/home">
<RouterLink :to={path:"/home"}> 
<RouterLink :to={name:"路由的名字"}> 
```

# 子路由


```ts
    routes:[

        {
	        name:"fisrtPage" //可以给路由命名
	        path:'/first',
	        component:Fisrt
	        children:[
		        {
			        path:xx 不用斜杠
			        name:yy
		        }
	        ]
        },

        {path:'/second',component:Second},

        {path:'/third',component:Third},

    ]
    <RouterLink :to={name:"路由的名字"}>  //这里需要全路径

```

	}
