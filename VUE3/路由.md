# 使用

- 创建`Router`文件夹，里面创建一个`index.ts`
- 引入`import { createRouter, createWebHistory } from "vue-router";`
- 导入`view`文件夹中的页面文件
- **定义路由表**
```ts
const routes = [

  { path: "/", component: Home },

  { path: "/setting", component: Setting },

];
```
- **创建路由对象**
```ts
const router = createRouter({
  history: createWebHistory(), // 使用 HTML5 模式
  routes
})

export default router
```
- **在主路由中挂载**
```ts
const app = createApp(App) app.use(router) // 👈 关键一步：插件挂载 app.mount('#app')
```
- **定义容器**
```vue
<router-view></router-view>
```
- 使用导航
```vue
<router-link to="/settings">设置</router-link>
```

在`views`或者`pages`中写组件，比如`home.vue、car.vue`
在router文件夹中，写index.ts
```ts
import {createRouter,createWebHistory} from 'vue-router'

import Fisrt from '../components/Fisrt.vue'

import Second from '../components/Second.vue'

// Make sure Third.vue exists in src/components, or correct the path if necessary

import Third from '../components/Third.vue'
  
const  router = createRouter({

    history:createWebHistory(),





})
export default router
```
创建路由
在app.vue中引入路由
```ts
<script setup lang="ts">

import { ref, computed } from 'vue'
import { RouterLink, RouterView } from 'vue-router'
 
</script>
 
<template>
  <div>
    <nav>
      <RouterLink to="/first">第一页</RouterLink>
      <RouterLink to="/second">这是第二页</RouterLink>
      <RouterLink to="/third">这是第三页</RouterLink>
    </nav>
    <RouterView />
  </div>
</template>
```

在**main.ts**中
```ts
import { createApp } from 'vue'
import App from './App.vue'
import router from './router'
const app = createApp(App)
app.use(router)
app.mount('#app')
```

# 工作模式
### history
不带#号
### hash
路径中有#号，服务端不用处理路径问题

# to

```ts
<RouterLink to="/home">
<RouterLink :to={path:"/home"}> 
<RouterLink :to={name:"路由的名字"}> 
```

# 路由参数
## query

```ts
<RouterLink :to="`/news/details?a=${news.id}&b=${news.name}`"}>  //a=haha b=baba就是参数
//&用于参数分割

//第二种方法
<RouterLink replace :to={path：'/news/details', //replace打开后不能后退
	query:{
	}

}>
```

使用参数
```ts
import {useRoute} from 'vue-router'
let route=useRoute()
route.query.参数名

```
## params
提前占位
```ts
{
	name:"fisrtPage" //可以给路由命名
	path:'/first',
	component:Fisrt
	children:[
		        {
			        path:'/news/:id/:name/:content'
			        name:yy
		        }
	        ]
        },
```


# 路由props
- 需要params参数
```ts
children:[
		        {
			        path:'/news/:id/:name/:content'
			        name:yy
			        props:true
		        }
	        ]
        },
        
 在其他文件拿到参数
 defineProps(['id','name'])
```
 - query
```ts
children:[
		        {
			        path:'/news/:id/:name/:content'
			        name:yy
			        props(route){
				        return route.query
			        }
		        }
	        ]
        },
```
