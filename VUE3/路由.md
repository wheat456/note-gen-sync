# 安装
`bun install vue-router` 首先要安装才能使用


# 使用

- 创建`Router`文件夹，里面创建一个`index.ts`
- 引入`import { createRouter, createWebHistory } from "vue-router";`
- 导入`view`文件夹中的页面文件
- **定义路由表**
```ts
const routes = [

  { path: "/", component: Home },

  { path: "/setting", component: Setting },

];
```
- **创建路由对象**
```ts
const router = createRouter({
  history: createWebHistory(), // 使用 HTML5 模式
  routes
})

export default router
```
- **在主路由中挂载**
```ts
const app = createApp(App) app.use(router) // 👈 关键一步：插件挂载 app.mount('#app')
```
- **定义容器**
```vue
<router-view></router-view>
```
- **使用导航**
```vue
<router-link to="/settings">设置</router-link>
```

# 工作模式
### history
不带#号
### hash
路径中有#号，服务端不用处理路径问题

# to
`to="xxx"` 这是一个死的字符串
`:to="xxx"` 可以将其作为变量使用
```ts
<RouterLink to="/home">
<RouterLink :to={path:"/home"}> 
<RouterLink :to={name:"路由的名字"}> 
```

# 子路由

- 使用**children**数组来接受一堆子路由
```ts
    routes:[

        {
	        name:"fisrtPage" //可以给路由命名
	        path:'/first',
	        component:Fisrt
	        children:[
		        {
			        path:xx 不用斜杠
			        name:yy
		        }
	        ]
        },

        {path:'/second',component:Second},

        {path:'/third',component:Third},

    ]
    <RouterLink :to={name:"路由的名字"}>  //这里需要全路径

```


# 并行路由
- 假如一个页面有两个`route-view`
- `<router-view name="helper"></router-view>` 起名
```ts
{
  path: '/dashboard',
  components: {
    default: MainDashboard, // 渲染到无名 <router-view>
    helper: SidePanel       // 渲染到 name="helper" 的 <router-view>
  }
}
```

# 路由参数
## query

```ts
<RouterLink :to="`/news/details?a=${news.id}&b=${news.name}`"}>  //a=haha b=baba就是参数
//&用于参数分割

//第二种方法
<RouterLink replace :to={path：'/news/details', //replace打开后不能后退
	query:{
	}

}>
```

使用参数
```ts
import {useRoute} from 'vue-router'
let route=useRoute()
route.query.参数名

```
## params
提前占位
```ts
{
	name:"fisrtPage" //可以给路由命名
	path:'/first',
	component:Fisrt
	children:[
		        {
			        path:'/news/:id/:name/:content'
			        name:yy
		        }
	        ]
        },
```


# 路由props
- 需要params参数
```ts
children:[
		        {
			        path:'/news/:id/:name/:content'
			        name:yy
			        props:true
		        }
	        ]
        },
        
 在其他文件拿到参数
 defineProps(['id','name'])
```
 - query
```ts
children:[
		        {
			        path:'/news/:id/:name/:content'
			        name:yy
			        props(route){
				        return route.query
			        }
		        }
	        ]
        },
```
